## Теоретическое обоснование

**Граф** — совокупность вершин и рёбер между ними. 
**Матрица смежности** — квадратная матрица V×V, где элемент `matrix[i][j] = 1`, если есть ребро из i в j. 
**DFS (Поиск в глубину)** — алгоритм обхода, идущий как можно дальше по каждому ребру, прежде чем вернуться назад. Реализуется рекурсивно или через стек.
**BFS (Поиск в ширину)** — алгоритм обхода, посещающий все соседние вершины перед переходом на следующий уровень. Реализуется через очередь. Используется для поиска кратчайшего пути в невзвешенном графе.

## Листинг программы

```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>

using namespace std;

// КЛАСС ГРАФА (Ориентированный, Матрица смежности)
class Graph {
    int V; // Количество вершин
    vector<vector<int>> matrix; // Матрица смежности

public:
    Graph(int vertices) : V(vertices), matrix(vertices, vector<int>(vertices, 0)) {}

    // Добавление ориентированного ребра
    void addEdge(int u, int v) {
        matrix[u][v] = 1;
    }

    // Добавление петли (ребро из вершины в саму себя)
    void addLoop(int u) {
        matrix[u][u] = 1;
    }

    // ОБХОД В ГЛУБИНУ (DFS)
    void DFS(int start) {
        vector<bool> visited(V, false);
        cout << "DFS from " << start << ": ";
        DFS_util(start, visited);
        cout << "\n";
    }

    // Рекурсивная вспомогательная функция для DFS
    void DFS_util(int v, vector<bool>& visited) {
        visited[v] = true;
        cout << v << " ";
        for (int u = 0; u < V; u++) {
            if (matrix[v][u] == 1 && !visited[u]) {
                DFS_util(u, visited);
            }
        }
    }

    // ОБХОД В ШИРИНУ (BFS)
    void BFS(int start) {
        vector<bool> visited(V, false);
        queue<int> q;
        visited[start] = true;
        q.push(start);
        cout << "BFS from " << start << ": ";
        
        while (!q.empty()) {
            int v = q.front(); q.pop();
            cout << v << " ";
            for (int u = 0; u < V; u++) {
                if (matrix[v][u] == 1 && !visited[u]) {
                    visited[u] = true;
                    q.push(u);
                }
            }
        }
        cout << "\n";
    }

    // ПОИСК КОМПОНЕНТ СВЯЗНОСТИ
    void printComponents() {
        vector<bool> visited(V, false);
        int compCount = 0;
        for (int i = 0; i < V; i++) {
            if (!visited[i]) {
                compCount++;
                cout << "Component " << compCount << ": ";
                DFS_util(i, visited);
                cout << "\n";
            }
        }
    }

    // КРАТЧАЙШИЙ ПУТЬ (BFS)
    vector<int> shortestPath(int start, int end) {
        vector<int> dist(V, -1);
        vector<int> prev(V, -1);
        queue<int> q;

        dist[start] = 0;
        q.push(start);

        // BFS для поиска кратчайшего расстояния
        while (!q.empty()) {
            int v = q.front(); q.pop();
            if (v == end) break;

            for (int u = 0; u < V; u++) {
                if (matrix[v][u] == 1 && dist[u] == -1) {
                    dist[u] = dist[v] + 1;
                    prev[u] = v;
                    q.push(u);
                }
            }
        }

        vector<int> path;
        if (dist[end] == -1) return path; // Пути нет

        // Восстановление пути с конца в начало
        for (int v = end; v != -1; v = prev[v]) {
            path.push_back(v);
        }
        reverse(path.begin(), path.end());
        return path;
    }

    void printShortestPath(int start, int end) {
        vector<int> path = shortestPath(start, end);
        if (path.empty()) {
            cout << "No path from " << start << " to " << end << "\n";
            return;
        }
        cout << "Shortest path " << start << " -> " << end << ": ";
        for (size_t i = 0; i < path.size(); i++) {
            cout << path[i];
            if (i < path.size() - 1) cout << " -> ";
        }
        cout << " (length: " << path.size() - 1 << ")\n";
    }

    // ДИАМЕТР ГРАФА (BFS дважды)
    int getDiameter() {
        if (V == 0) return 0;
        
        // 1-й BFS: находим самую удаленную вершину (farthestNode) от вершины 0
        vector<int> dist1(V, -1);
        queue<int> q;
        dist1[0] = 0;
        q.push(0);
        int farthestNode = 0;
        
        while (!q.empty()) {
            int v = q.front(); q.pop();
            if (dist1[v] > dist1[farthestNode]) farthestNode = v;
            
            for (int u = 0; u < V; u++) {
                if (matrix[v][u] == 1 && dist1[u] == -1) {
                    dist1[u] = dist1[v] + 1;
                    q.push(u);
                }
            }
        }
        
        // 2-й BFS: от farthestNode находим максимальное расстояние до других вершин
        vector<int> dist2(V, -1);
        dist2[farthestNode] = 0;
        q.push(farthestNode);
        int diameter = 0;
        
        while (!q.empty()) {
            int v = q.front(); q.pop();
            if (dist2[v] > diameter) diameter = dist2[v];
            
            for (int u = 0; u < V; u++) {
                if (matrix[v][u] == 1 && dist2[u] == -1) {
                    dist2[u] = dist2[v] + 1;
                    q.push(u);
                }
            }
        }
        
        return diameter;
    }
};

int main() {
    cout << "--- Part 1 & 2: Graph, DFS, BFS (Variant 12: Directed with Loops) ---\n";
    Graph g(6);
    
    // Добавляем ребра (ориентированные)
    g.addEdge(0, 1);
    g.addEdge(0, 3);
    g.addEdge(1, 2);
    g.addEdge(2, 4);
    g.addEdge(3, 5);
    g.addEdge(4, 5);
    
    // Добавляем петли (специфика 12 варианта)
    g.addLoop(0);
    g.addLoop(3);

    g.DFS(0);
    g.BFS(0);

    cout << "\n--- Part 3: Components & Shortest Path ---\n";
    g.printComponents();
    g.printShortestPath(0, 5);

    cout << "\n--- Part 2 (Variant 12): Graph Diameter ---\n";
    cout << "Graph diameter: " << g.getDiameter() << "\n";

    return 0;
}
```

## Сравнительный анализ

| Алгоритм | Время (V - вершин, E - рёбер) | Память | Особенности |
| :--- | :--- | :--- | :--- |
| **Матрица смежности** | O(V²) на обход | O(V²) | Быстрая проверка наличия ребра O(1). Хороша для плотных графов. |
| **Список смежности** | O(V + E) на обход | O(V + E) | Экономит память на разреженных графах. Проверка ребра O(degree). |
| **DFS** | O(V + E) | O(V) | Идеален для поиска циклов, компонент, топологической сортировки. |
| **BFS** | O(V + E) | O(V) | Идеален для кратчайшего пути в невзвешенном графе. |

## Контрольные вопросы

1. **Что такое граф? Из каких элементов он состоит?**
   Граф — это математическая структура, состоящая из множества вершин (узлов) V и множества рёбер (связей) E, соединяющих пары вершин.

2. **Чем отличается ориентированный граф от неориентированного?**
   В ориентированном графе рёбра имеют направление (из вершины A в вершину B), то есть связь односторонняя. В неориентированном графе рёбра не имеют направления (связь двусторонняя, A-B).

3. **Что такое матрица смежности? Каковы её достоинства и недостатки?**
   Это квадратная матрица размером V×V, где `matrix[i][j] = 1`, если есть ребро из i в j, и 0 — если нет.
   *Достоинства:* Проверка наличия ребра за O(1); простота реализации.
   *Недостатки:* Занимает O(V²) памяти, что крайне неэффективно для разреженных графов (где мало рёбер).

4. **Что такое список смежности? Каковы его достоинства и недостатки?**
   Это массив (или вектор) списков, где для каждой вершины хранится список её соседей.
   *Достоинства:* Экономия памяти O(V + E), что идеально для разреженных графов.
   *Недостатки:* Проверка наличия конкретного ребра занимает O(degree) (нужно перебрать список соседей).

5. **В каких случаях лучше использовать матрицу смежности, а в каких - список смежности?**
   *Матрицу* лучше использовать для плотных графов (где количество рёбер E близко к V²), когда нужно быстро (за O(1)) проверять наличие конкретных связей. *Список* — для разреженных графов (E << V²) и когда нужно быстро перебрать всех соседей вершины.

6. **Какова сложность проверки наличия ребра в матрице смежности? В списке смежности?**
   В матрице смежности — O(1). В списке смежности — O(degree(v)), где degree(v) — степень вершины (количество её соседей).

7. **Какова сложность перебора всех соседей вершины в матрице смежности? В списке смежности?**
   В матрице смежности — O(V), так как нужно пройти по всей строке матрицы. В списке смежности — O(degree(v)), так как мы просто идем по списку соседей.

8. **Что такое поиск в глубину (DFS)? Как он работает?**
   Это алгоритм обхода графа. Работает по принципу «иди как можно дальше, пока можешь»: заходит в первую непосещенную смежную вершину, затем в её соседа и так до тупика, после чего возвращается назад (backtracking) и пробует другие пути.

9. **Какие задачи решаются с помощью DFS?**
   Поиск компонент связности, проверка наличия циклов, топологическая сортировка, поиск мостов и точек сочленения, генерация лабиринтов.

10. **Что такое поиск в ширину (BFS)? Как он работает?**
    Это алгоритм обхода графа. Работает по уровням: сначала посещает всех прямых соседей стартовой вершины, затем соседей соседей (следующий уровень) и так далее. Реализуется с помощью очереди.

11. **Какие задачи решаются с помощью BFS?**
    Поиск кратчайшего пути в невзвешенном графе, поиск всех вершин на заданном расстоянии, проверка двудольности графа, поиск диаметра графа.

12. **В чём отличие рекурсивной реализации DFS от итеративной?**
    Рекурсивная использует неявный стек вызовов функций (память O(V)), что может привести к переполнению стека на очень больших графах. Итеративная использует явную структуру данных `std::stack` и цикл, что безопаснее для глубоких графов и дает контроль над памятью.

13. **Почему BFS находит кратчайший путь в невзвешенном графе, а DFS - нет?**
    BFS исследует граф строго по уровням (сначала все вершины на расстоянии 1, потом на расстоянии 2 и т.д.), поэтому первым делом он достигает цели именно по кратчайшему пути. DFS же может уйти в длинную «ветку» и найти путь, который будет длиннее, даже если есть короткий.

14. **Как найти компоненты связности графа с помощью DFS?**
    Нужно запустить цикл по всем вершинам. Если вершина не посещена — запускаем из неё DFS (или BFS), который обойдет всю её компоненту связности. Счетчик запусков и будет количеством компонент.

15. **Как проверить наличие цикла в графе с помощью DFS?**
    При обходе в глубину нужно отслеживать не только посещенные вершины, но и текущий «путь» (стек рекурсии). Если при обходе мы встречаем вершину, которая уже находится в текущем стеке рекурсии (а не просто посещена ранее), значит, есть цикл.
